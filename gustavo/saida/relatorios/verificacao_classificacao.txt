        ^ TESTE: só a transição cujo alvo é o ano de teste
            p = {("B1R_lag", 0): te[f"ABANDONO_{s}_L1"].to_numpy(),
        ^ régua B1R_lag: abandono de t−1, sem ajuste nenhum

        ^ régua B2: mediana calculada só no treino
            for k in range(1, N_BLOCOS + 1):
        ^ blocos cumulativos 1, 2 e 3
                cols = colunas_bloco(etapa, k)

                for nome in MODELOS:

                    pipe = modelo(nome, cols).fit(tr[cols], tr[col_alvo].to_numpy())
